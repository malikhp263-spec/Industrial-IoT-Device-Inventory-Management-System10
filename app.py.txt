# ======================
# Threats
# ======================

elif menu == "Threats":

    st.title("Cyber Threat Landscape")

    threat_data = pd.DataFrame({
        "Threat":["Malware","Ransomware","DDoS","Phishing","Insider"],
        "Risk":[90,85,70,60,50]
    })

    fig = px.pie(
        threat_data,
        values="Risk",
        names="Threat",
        hole=0.5,
        title="Threat Distribution"
    )

    st.plotly_chart(fig, use_container_width=True)

    st.info("""
    Industrial IoT environments face multiple security risks including:
    • Malware Infections
    • Ransomware Attacks
    • Denial of Service Attacks
    • Insider Threats

# ======================
# Risk Analysis
# ======================

elif menu == "Risk Analysis":

    st.title("Risk Assessment")

    risk = st.slider(
        "Current Security Risk Level",
        0,100,65
    )

    fig = go.Figure(go.Indicator(
        mode="gauge+number",
        value=risk,
        title={'text':"Risk Score"},
        gauge={
            'axis':{'range':[0,100]},
            'bar':{'color':"red"},
            'steps':[
                {'range':[0,40],'color':"green"},
                {'range':[40,70],'color':"yellow"},
                {'range':[70,100],'color':"red"}
            ]
        }
    ))

    st.plotly_chart(fig, use_container_width=True)

    if risk < 40:
        st.success("Low Risk")
    elif risk < 70:
        st.warning("Medium Risk")
    else:
        st.error("High Risk")
